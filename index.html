            const osc = ctx.createOscillator();

            const gain = ctx.createGain();

            osc.connect(gain);

    function animateConfetti() {

        cctx.clearRect(0, 0, confettiCanvas.width, confettiCanvas.height);

        confettiParticles.forEach(p => {

            p.x += p.vx; p.y += p.vy; p.vy += 0.05; p.rot += p.vr; p.life++;

            cctx.save();

            cctx.translate(p.x, p.y);

            cctx.rotate(p.rot * Math.PI / 180);

            cctx.fillStyle = p.color;

            cctx.globalAlpha = Math.max(0, 1 - p.life/150);

            if (p.shape === 'circle') {

                cctx.beginPath();

                cctx.arc(0, 0, p.size/2, 0, 7);

                cctx.fill();

            } else {

                cctx.fillRect(-p.size/2, -p.size/2, p.size, p.size*0.6);

            }

            cctx.restore();

        });

        confettiParticles = confettiParticles.filter(p => p.life < 150 && p.y < confettiCanvas.height + 40);

        if (confettiParticles.length > 0) {

            confettiRAF = requestAnimationFrame(animateConfetti);

        } else {

            confettiRAF = null;

            cctx.clearRect(0, 0, confettiCanvas.width, confettiCanvas.height);

        }

    }

    function produzirProduto(key) {

        const prod = PRODUTOS.find(p => p.key === key);

        if (!prod) return;

        for (let ins in prod.insumos) {

            const needed = prod.insumos[ins];

            let disponivel = (state.insumos[ins]||0);

            if (disponivel < needed && INSUMOS_PRODUTO.includes(ins)) {

                disponivel += (state.produtos[ins]||0);

            }

            if (disponivel < needed) {

                toast(`⚠️ Falta ${ins} (precisa de ${needed}, tem ${disponivel})`);

                return;

            }

        }

        if (state.maoDeObra < prod.mao_obra) {

            toast(`🧑‍🌾 Mão de obra insuficiente (precisa de ${prod.mao_obra}, tem ${state.maoDeObra})`);

            return;

        }

        // Consome

        for (let ins in prod.insumos) {

            let needed = prod.insumos[ins];

            let fromInsumo = Math.min(state.insumos[ins]||0, needed);

            if (fromInsumo > 0) {

                consumirInsumo(ins, fromInsumo, 'Consumiu para '+prod.label);

                needed -= fromInsumo;

            }

            if (needed > 0 && INSUMOS_PRODUTO.includes(ins)) {

                let fromProduto = Math.min(state.produtos[ins]||0, needed);

                if (fromProduto > 0) {

                    state.produtos[ins] -= fromProduto;

                    pushNotificacao(`📦 -${fromProduto} ${ins} (produto) consumido`, 'consumo');

                    needed -= fromProduto;

                }

            }

            if (needed > 0) {

                toast(`⚠️ Erro ao consumir ${ins}, necessário ${prod.insumos[ins]}`);

                return;

            }

        }

        state.maoDeObra -= prod.mao_obra;

        adicionarProduto(key, 1, 'Produziu '+prod.label);

        updateWeekly('produtos_produzidos', 1);

        if (key === 'queijo') state.semanaMetrics.produtos_queijo = (state.semanaMetrics.produtos_queijo||0) + 1;

        checkAll();

        saveState();

        renderAll();

        toast('🧀 '+prod.label+' produzido!');

        launchConfetti(30);

        playBeep(700, 100);

    }

    function fmtDate(ts) {

        const d = new Date(ts);

        return d.toLocaleDateString('pt-BR') + ' ' + d.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' });

    }

    function drawChart() {

        const ctx = document.getElementById('weekChart');

        if (!ctx || typeof Chart === 'undefined') return;

        const days = [];

        const counts = [];

        const now = new Date();

        for (let i=6; i>=0; i--) {

            const d = new Date(now);

            d.setDate(now.getDate() - i);

            const key = d.toISOString().slice(0,10);

            days.push(d.toLocaleDateString('pt-BR', { weekday: 'short' }));

            const diario = state.pomodorosDiarios.find(item => item.data === key);

            counts.push(diario ? diario.count : 0);

        }

        if (chartInstance) chartInstance.destroy();

        chartInstance = new Chart(ctx, {

            type: 'bar',

            data: {

                labels: days,

                datasets: [{

                    label: 'Pomodoros',

                    data: counts,

                    backgroundColor: getComputedStyle(document.documentElement).getPropertyValue('--sage').trim(),

                    borderRadius: 6

                }]

            },

            options: {

                plugins: { legend: { display: false } },

                scales: { y: { beginAtZero: true, ticks: { stepSize: 1 } } }

            }

        });

    }

    function downloadFile(filename, content, type) {

        const blob = new Blob([content], { type });

        const url = URL.createObjectURL(blob);

        const a = document.createElement('a');

        a.href = url; a.download = filename; a.click();

        URL.revokeObjectURL(url);

    }

    function importBackup(file) {

        const reader = new FileReader();

        reader.onload = e => {

            try {

                const parsed = JSON.parse(e.target.result);

                if (typeof parsed.pomodoros === 'undefined' || typeof parsed.graos === 'undefined') {

                    toast('⚠️ Arquivo inválido.');

                    return;

                }

                state = Object.assign(defaultState(), parsed);

                if (!state.tasksDone || state.tasksDone.length !== TOTAL_TASKS) {

                    state.tasksDone = new Array(TOTAL_TASKS).fill(false);

                }

                saveState();

                applyTheme();

                renderAll();

                toast('✅ Backup importado!');

            } catch (err) {

                toast('⚠️ Erro ao ler backup.');

            }

        };

        reader.readAsText(file);

    }
